=======================================================
ORG-SERVICE (Port 8081) — Auth : JWT WorkHub (plus de Clerk)
=======================================================
// Ajouter dans tous les appels protégés :
// Header: Authorization: Bearer <JWT_TOKEN>
// (JWT obtenu depuis identity-service : POST http://localhost:8088/api/auth/login)

// 1. Créer une Organisation
POST http://localhost:8081/api/orgs
Authorization: Bearer <JWT_TOKEN>
{
  "name": "WorkHub Corp",
  "legalName": "WorkHub International S.A.S",
  "city": "Paris"
}

// 2. Lister les Orgs (Récupérer l'ID)
GET http://localhost:8081/api/orgs
Authorization: Bearer <JWT_TOKEN>

// 3. Créer un Département
POST http://localhost:8081/api/departments
Authorization: Bearer <JWT_TOKEN>
{
  "organizationId": "VOTRE_ID_ORG",
  "name": "Ressources Humaines",
  "description": "Gestion des talents"
}

// 4. Créer un Poste (Position)
POST http://localhost:8081/api/positions
Authorization: Bearer <JWT_TOKEN>
{
  "organizationId": "VOTRE_ID_ORG",
  "title": "Développeur Fullstack",
  "description": "Java / React",
  "category": "CADRE"
}

// 5. Voir les départements d'une Org
GET http://localhost:8081/api/orgs/VOTRE_ID_ORG/departments
Authorization: Bearer <JWT_TOKEN>

// 6. Voir les postes d'une Org
GET http://localhost:8081/api/orgs/VOTRE_ID_ORG/positions
Authorization: Bearer <JWT_TOKEN>

// 7. Dashboard d'une Org
GET http://localhost:8081/api/orgs/VOTRE_ID_ORG/dashboard
Authorization: Bearer <JWT_TOKEN>

// 8. Routes INTERNES (sans auth, inter-services uniquement)
GET http://localhost:8081/internal/orgs/VOTRE_ID_ORG/exists
GET http://localhost:8081/internal/departments/VOTRE_ID_DEPT/exists
GET http://localhost:8081/internal/departments/VOTRE_ID_DEPT/org

=======================================================
IDENTITY SERVICE (Port 8088)
=======================================================

// 1. S'inscrire (créer un compte + obtenir un JWT)
POST http://localhost:8088/api/auth/register
{
  "organizationId": "VOTRE_ID_ORG",
  "email": "jean.dupont@exemple.com",
  "password": "motdepasse",
  "firstName": "Jean",
  "lastName": "Dupont",
  "role": "EMPLOYEE"
}

// 2. Se connecter (obtenir un JWT)
POST http://localhost:8088/api/auth/login
{
  "email": "jean.dupont@exemple.com",
  "password": "motdepasse"
}
// => Réponse : { "token": "eyJ..." }

// 3. Lister les utilisateurs (JWT requis)
GET http://localhost:8088/api/users
Authorization: Bearer <JWT_TOKEN>

// 4. Créer un utilisateur directement
POST http://localhost:8088/api/users
Authorization: Bearer <JWT_TOKEN>
{
  "organizationId": "VOTRE_ID_ORG",
  "email": "marie.martin@exemple.com",
  "password": "motdepasse",
  "firstName": "Marie",
  "lastName": "Martin",
  "role": "ORG_ADMIN"
}
