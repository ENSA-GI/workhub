=======================================================
IDENTITY SERVICE (Port 8088) — Auth JWT WorkHub (plus de Clerk/Svix)
=======================================================

// 1. S'inscrire (aucun token requis)
POST http://localhost:8088/api/auth/register
Content-Type: application/json
{
  "organizationId": "VOTRE_ID_ORG",
  "email": "jean.dupont@exemple.com",
  "password": "motdepasse123",
  "firstName": "Jean",
  "lastName": "Dupont",
  "role": "EMPLOYEE"
}

// 2. Se connecter → récupérer le JWT
POST http://localhost:8088/api/auth/login
Content-Type: application/json
{
  "email": "jean.dupont@exemple.com",
  "password": "motdepasse123"
}
// Réponse : { "token": "eyJhbGciOiJIUzI1NiJ9..." }

// 3. Lister les utilisateurs (JWT requis)
GET http://localhost:8088/api/users
Authorization: Bearer <JWT_TOKEN>

// 4. Créer un utilisateur (JWT requis)
POST http://localhost:8088/api/users
Authorization: Bearer <JWT_TOKEN>
Content-Type: application/json
{
  "organizationId": "VOTRE_ID_ORG",
  "email": "marie.martin@exemple.com",
  "password": "motdepasse123",
  "firstName": "Marie",
  "lastName": "Martin",
  "role": "ORG_ADMIN"
}

// 5. Désactiver un utilisateur
PATCH http://localhost:8088/api/users/VOTRE_USER_ID/deactivate
Authorization: Bearer <JWT_TOKEN>

// 6. Health check (sans token)
GET http://localhost:8088/actuator/health
